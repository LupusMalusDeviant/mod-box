name: CI

on:
  push:
    branches: [main]
  pull_request:

permissions:
  contents: read

jobs:
  check:
    name: Validate and smoke-test (${{ matrix.os }})
    runs-on: ${{ matrix.os }}
    timeout-minutes: 30
    strategy:
      fail-fast: false
      matrix:
        os: [ubuntu-latest, windows-latest]
    steps:
      - uses: actions/checkout@v7

      - uses: actions/setup-node@v7
        with:
          node-version: 26

      # A download on the runner can hang: each try gets 4 minutes, a hung one is tried once more.
      - name: Install Claude Code
        shell: bash
        timeout-minutes: 10
        run: |
          for try in 1 2; do
            timeout 240 npm install -g @anthropic-ai/claude-code && exit 0
            echo "::warning::npm install failed or hung (try $try)"
          done
          exit 1

      - name: Validate the marketplace
        run: claude plugin validate .

      - name: Validate and test every plugin
        shell: bash
        run: |
          for p in plugins/*/; do
            echo "== $p"
            claude plugin validate "$p"
            claude plugin test "$p"
          done
